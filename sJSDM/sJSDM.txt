plotFuncMAGs <- function( func, taxRank, graphYmax){
                  whichFuncMAGs <- asvMAG[asvMAG[func] == 1,"MAG_ID"] 
                  PS_magfunc <- prune_taxa(tax_table(PS_mag)[,"Species"] %in% whichFuncMAGs, PS_mag)
                  ps.justPalsa <- subset_samples(PS_magfunc, Subsite=="Palsa")
                  ps.justThermokarst <- subset_samples(PS_magfunc, Subsite=="Thermokarst")
                  funcMAG.Pal <- plot_bar(ps.justPalsa, x="DepthSection", fill = taxRank, facet_grid=~Core ) +
                     geom_bar(aes(color=get(taxRank), fill=get(taxRank)), stat="identity", position="stack") +
                     theme(axis.text=element_text(size=12)) +
                     theme( 
                     #       axis.text.y=element_blank(),
                     #       axis.ticks.y=element_blank(),
                            legend.position="none",
                            ) +
                     labs(title='Palsa') + theme(plot.title = element_text(hjust = 0.5))
                  funcMAG.tk  <- plot_bar(ps.justThermokarst, x="DepthSection", fill = taxRank, facet_grid=~Core ) +
                     geom_bar(aes(color=get(taxRank), fill=get(taxRank)), stat="identity", position="stack") +
                     theme(axis.text=element_text(size=12)) +
                     theme( 
                            axis.text.y=element_blank(),
                            axis.ticks.y=element_blank(),
                            legend.text=element_text(size=20)
                           ) +
                           guides(color="none") +
                     ylab(NULL) +
                     labs(title='Thermokarst') + theme(plot.title = element_text(hjust = 0.5))
                  funcPlot <- funcMAG.Pal + funcMAG.tk +
                                   plot_annotation(title=func, 
                                          theme = theme(plot.title = element_text(size = 18, hjust=0.5)))
                  funcPlot & ylim(min(0), max(graphYmax))
                  }

Occ <- community$response

SP <- matrix(rnorm(200, 0, 0.3), 100, 2)

model <- sjSDM(Y = Occ, 
               env = linear(data = Env, formula = ~X1+X2+X3), 
               spatial = linear(data = SP, formula = ~0+X1:X2), 
               se = TRUE, family=binomial("probit"), 
               sampling = 100L, 
               verbose = FALSE)

summary(model)

plot(model)

image(getCor(model))

an = anova(model, verbose = FALSE)

plot(an)

results = internalStructure(an)

plot(results)
