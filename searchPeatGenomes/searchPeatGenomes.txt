## function for getting all 16S matches out of one of these:
def pullOut16S(barrnapFile):
    allSeqsOneBarrnap = [seq_record for seq_record in SeqIO.parse(barrnapFile, "fasta")]
    recs16S_i = [ i for i in allSeqsOneBarrnap if "16S_rRNA" in i.id ]
    genomeNCBInum = "_".join(barrnapFile.split("_")[0:2])
    bactOrArch = barrnapFile.split("_")[2].replace("rRNA.fna", "")
    for i,obj in enumerate(recs16S_i): 
        obj.id = genomeNCBInum+"."+bactOrArch+"."+str(i); 
        #obj.name = obj.name + " " + bactOrArch 
    return(recs16S_i)
